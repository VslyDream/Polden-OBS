cmake_minimum_required(VERSION 3.28)
project(polden-file-transfer-tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 REQUIRED COMPONENTS Core)
add_executable(polden-file-transfer-tests test-file-transfer.cpp ../../frontend/utility/PoldenFileTransfer.cpp)
target_link_libraries(polden-file-transfer-tests PRIVATE Qt6::Core)
add_executable(polden-slow-converter slow-converter.cpp)
target_link_libraries(polden-slow-converter PRIVATE Qt6::Core)
if(MSVC)
  target_compile_options(polden-file-transfer-tests PRIVATE /utf-8)
endif()
enable_testing()
add_test(NAME polden-file-transfer COMMAND polden-file-transfer-tests)
add_executable(polden-default-preset-tests test-default-preset.cpp ../../frontend/utility/PoldenDefaults.cpp)
target_link_libraries(polden-default-preset-tests PRIVATE Qt6::Core)
add_test(NAME polden-default-preset COMMAND polden-default-preset-tests "${CMAKE_CURRENT_SOURCE_DIR}/../../frontend/data/polden/defaults")
add_executable(polden-redesign-tests test-redesign.cpp ../../frontend/utility/PoldenTemplates.cpp)
target_link_libraries(polden-redesign-tests PRIVATE Qt6::Core)
if(MSVC)
  target_compile_options(polden-redesign-tests PRIVATE /utf-8)
endif()
add_test(NAME polden-redesign COMMAND polden-redesign-tests)
